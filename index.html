function buildLines(overall, rank){ // ===== 静穏ゾーン ===== if(rank === "静穏"){ const seion = 100 - Math.abs(overall - 50) * 2; // 45〜55% のときだけ静穏運を表示 if(overall >= 45 && overall <= 55){ // ★50%は特別扱い if(overall === 50){ return [ "静穏運 100%", "Perfectly Average", "", "今日は派手な事件は起きない。安心。", "何も壊れていない日。" ].join("\n"); } return [ `静穏運 ${seion}%`, "", "今日は普通。", "特別ではないが問題もない。" ].join("\n"); } // 40〜44 / 56〜59 は普通の静穏 return [ "今日は普通。", "特別ではないが問題もない。" ].join("\n"); } // ===== 他ランク ===== const MAP = { "絶不調": [ "今日は運勢が良くない。", "どうせ外れるから気にするな。" ], "低調": [ "今日は少し流れが悪い。", "まあ、そんな日もある。" ], "好調": [ "今日は悪くない。", "少しだけ動いてもいいかもしれない。" ], "絶好調": [ "今日はかなり良い流れ。", "珍しい日だ。" ] }; return (MAP[rank] || ["今日はそんな日。"]).join("\n"); }
